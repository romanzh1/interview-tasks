func appendSlice(sl []int, val int) {
	sl = append(sl, val)
}


func mutateSlice(sl []int, idx, val int) {
	sl[idx] = val
}

type Order struct {
    Product  string
    Status   string
    ID       int64
    Quantity int
}

func SaveOrder(ctx context.Context, db *sql.DB, order Order) (Order, error) {
    if err := db.QueryRowContext(ctx,
        "INSERT INTO orders(product, quantity, status) VALUES($1, $2, $3) RETURNING id",
        order.Product, order.Quantity, order.Status).Scan(&order.ID); err!=nil{
          return nil, err
        }

    return order, nil
}


func SendOrderCreatedEvent(order Order) error {
    // пишет в кафку
    fmt.Printf("Отправка события OrderCreated: %+v\n", order)
    return nil
}

func main() {
    ctx := context.Background()

    db, err := sql.Open("postgres", "user=postgres dbname=testdb sslmode=disable")
    if err != nil {
        log.Fatal(err)
    }
    defer db.Close()

    order := Order{
        Product:  "Book",
        Quantity: 1,
        Status:   "NEW",
    }



    savedOrder, err := SaveOrder(ctx, db, order)
    if err != nil {
        log.Fatal("Ошибка сохранения заказа:", err)
    }
    fmt.Println("Заказ сохранён с ID:", savedOrder.ID)

    if err := SendOrderCreatedEvent(savedOrder); err != nil {
        log.Println("Ошибка отправки события в брокер:", err)
    }
}

func proccess(abc []string) map[string][]int{
  m := make(map[string][]int)

  mu := sync.Mutex{}
  wg := sync.WaitGroup()
  sem := NewSemaphore(5)

  wg.Add(len(abc))
  for i, el := range abc{
    sem.Acquire()
    go func(){
      mu.Lock()
      if mel, ok := m[el]{
        m[el] = append(m[el], i)
      } else{
        m[el] = make([]int, 1)
        m[el][0] = i
      }
      mu.Unlock()
      wg.Done()
      sem.Release()
    }()
  }

  wg.Wait()

  return m
}

type struct semaphore {
  chan struct{} ch
}

func NewSemaphore(thr int) semaphore {
  return semaphore{
    ch: make(chan struct{}, thr)
  }
}

func (s *semaphore) Acquire(){
  s.ch <- struct{}
}

func (s *semaphore) Release(){
  <-s.ch
}
